const esCorreoRealValido = (correo) => {
  const limpio = (correo || '').trim().toLowerCase();
  const match = limpio.match(/^[^\s@]+@([^\s@]+\.[^\s@]{2,})$/);
  if (!match) return false;
  const dominio = match[1];
  if (DOMINIOS_CORREO_FALSO.includes(dominio)) return false;
  return true;
};

const leerCuentasComerciantes = () => {
  try {
    const raw = JSON.parse(localStorage.getItem(COMERCIANTES_CUENTAS_KEY) || '{}');
    const normalizadas = {};
    Object.entries(raw).forEach(([correo, val]) => {
      if (typeof val === 'string') {
        normalizadas[correo] = {
          password: val,
          aprobado: true,
          creadoEn: new Date().toISOString()
        };
      } else if (val && typeof val === 'object') {
        normalizadas[correo] = {
          password: val.password || '',
          aprobado: Boolean(val.aprobado),
          creadoEn: val.creadoEn || new Date().toISOString()
        };
      }
    });
    return normalizadas;
  } catch {
    return {};
  }
};

const guardarCuentasComerciantes = (cuentas) => {
  localStorage.setItem(COMERCIANTES_CUENTAS_KEY, JSON.stringify(cuentas));
};

          <section className="bg-white rounded-2xl border border-amber-300 p-4 sm:p-5 space-y-4 shadow-sm">
            <div className="flex items-center justify-between border-b border-neutral-100 pb-3">
              <div>
                <span className="text-[11px] font-bold uppercase tracking-wider text-amber-700 bg-amber-50 px-2.5 py-0.5 rounded-full">
                  Gestión de Local
                </span>
                <h2 className="text-base sm:text-lg font-extrabold text-neutral-900 mt-1">
                  Panel de Comerciante
                </h2>
              </div>
              <button
                type="button"
                onClick={() => setPanelActivo(null)}
                className="text-xs font-bold text-neutral-500 hover:text-neutral-800 px-2.5 py-1 rounded-lg bg-neutral-100"
              >
                Cerrar
              </button>
            </div>

            <div className="bg-amber-50/60 border border-amber-200 rounded-xl p-4 space-y-3">
              <div className="flex items-center justify-between gap-2">
                <h3 className="text-sm font-extrabold text-neutral-900">
                  {modoRecuperarComerciante
                    ? 'Recuperar contraseña de comerciante'
                    : modoRegistroComerciante
                      ? 'Crear cuenta única de comerciante'
                      : 'Ingresar a mi cuenta de comerciante'}
                </h3>
                {comercianteAutenticado && (
                  <button
                    type="button"
                    onClick={() => {
                      setComercianteAutenticado('');
                      setPasswordComerciante('');
                      setMensajeAuthComerciante('Sesión de comerciante cerrada.');
                      setErrorAuthComerciante('');
                    }}
                    className="text-xs font-bold text-red-600 hover:underline"
                  >
                    Cerrar sesión
                  </button>
                )}
              </div>

              {!comercianteAutenticado && !modoRecuperarComerciante && (
                <div className="grid grid-cols-2 gap-1.5 bg-white p-1 rounded-xl border border-amber-200">
                  <button
                    type="button"
                    onClick={() => {
                      setModoRegistroComerciante(false);
                      setErrorAuthComerciante('');
                      setMensajeAuthComerciante('');
                      setIntentosFallidosComerciante(0);
                    }}
                    className={`py-1.5 text-xs font-bold rounded-lg transition ${
                      !modoRegistroComerciante
                        ? 'bg-orange-600 text-white'
                        : 'text-neutral-700 hover:bg-neutral-50'
                    }`}
                  >
                    Ya tengo cuenta
                  </button>
                  <button
                    type="button"
                    onClick={() => {
                      setModoRegistroComerciante(true);
                      setErrorAuthComerciante('');
                      setMensajeAuthComerciante('');
                      setIntentosFallidosComerciante(0);
                    }}
                    className={`py-1.5 text-xs font-bold rounded-lg transition ${
                      modoRegistroComerciante
                        ? 'bg-orange-600 text-white'
                        : 'text-neutral-700 hover:bg-neutral-50'
                    }`}
                  >
                    Crear cuenta nueva
                  </button>
                </div>
              )}

              {!comercianteAutenticado ? (
                <form
                  onSubmit={(e) => {
                    e.preventDefault();
                    setErrorAuthComerciante('');
                    setMensajeAuthComerciante('');
                    const correo = emailComerciante.trim().toLowerCase();
                    if (!correo) return;

                    if (correo === ADMIN_EMAIL_UNICO) {
                      setErrorAuthComerciante(
                        'Este correo corresponde al administrador. Ingresá desde el Panel de Administrador.'
                      );
                      return;
                    }

                    if (!esCorreoRealValido(correo)) {
                      setErrorAuthComerciante(
                        'Ingresá un correo electrónico real y válido (ej.: tunombre@gmail.com).'
                      );
                      return;
                    }

                    const cuentas = leerCuentasComerciantes();

                    if (modoRecuperarComerciante) {
                      if (!cuentas[correo]) {
                        setErrorAuthComerciante(
                          'No existe una cuenta registrada con ese correo de comerciante.'
                        );
                        return;
                      }
                      if (!nuevaPasswordComerciante.trim() || nuevaPasswordComerciante.trim().length < 4) {
                        setErrorAuthComerciante(
                          'Ingresá una nueva contraseña única de al menos 4 caracteres.'
                        );
                        return;
                      }
                      cuentas[correo] = {
                        ...cuentas[correo],
                        password: nuevaPasswordComerciante
                      };
                      guardarCuentasComerciantes(cuentas);
                      setCuentasComerciantes(cuentas);
                      setNuevaPasswordComerciante('');
                      setPasswordComerciante('');
                      setModoRecuperarComerciante(false);
                      setModoRegistroComerciante(false);
                      setMensajeAuthComerciante(
                        `Contraseña actualizada para ${correo}. Ya podés iniciar sesión con tu nueva clave.`
                      );
                      return;
                    }

                    if (!passwordComerciante.trim() || passwordComerciante.trim().length < 4) {
                      setErrorAuthComerciante(
                        'Ingresá tu contraseña única (mínimo 4 caracteres).'
                      );
                      return;
                    }

                    if (modoRegistroComerciante) {
                      if (cuentas[correo]) {
                        setErrorAuthComerciante(
                          'Ya existe una cuenta registrada con este correo. Cambiá a "Ya tengo cuenta" para ingresar.'
                        );
                        return;
                      }
                      cuentas[correo] = {
                        password: passwordComerciante,
                        aprobado: false,
                        creadoEn: new Date().toISOString()
                      };
                      guardarCuentasComerciantes(cuentas);
                      setCuentasComerciantes(cuentas);
                      setPasswordComerciante('');
                      setModoRegistroComerciante(false);
                      setMensajeAuthComerciante(
                        `Cuenta creada para ${correo}. Quedó pendiente de aprobación por el administrador antes de poder ingresar.`
                      );
                      return;
                    }

                    const cuenta = cuentas[correo];
                    if (!cuenta) {
                      setErrorAuthComerciante(
                        'Todavía no creaste una cuenta con este correo. Tocá "Crear cuenta nueva" para registrarte.'
                      );
                      return;
                    }

                    if (cuenta.password !== passwordComerciante) {
                      const nuevoIntento = intentosFallidosComerciante + 1;
                      setIntentosFallidosComerciante(nuevoIntento);
                      setErrorAuthComerciante(
                        `Contraseña incorrecta (intento ${nuevoIntento}). Podés volver a intentarlo las veces que quieras o elegir "¿Olvidaste tu contraseña? Recuperarla".`
                      );
                      return;
                    }

                    if (!cuenta.aprobado) {
                      setIntentosFallidosComerciante(0);
                      setErrorAuthComerciante(
                        `Tu cuenta (${correo}) ya fue creada, pero todavía está pendiente de aprobación por el administrador.`
                      );
                      return;
                    }

                    setIntentosFallidosComerciante(0);
                    setComercianteAutenticado(correo);
                    setMensajeAuthComerciante(`Sesión iniciada como comerciante aprobado (${correo}).`);
                  }}
                  className="space-y-3"
                >
                  <div>
                    <label className="block text-xs font-bold text-neutral-700 mb-1">
                      Correo electrónico real
                    </label>
                    <input
                      type="email"
                      required
                      value={emailComerciante}
                      onChange={(e) => setEmailComerciante(e.target.value)}
                      placeholder="tucomercio@gmail.com"
                      className="w-full rounded-xl border border-neutral-200 bg-white px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
                    />
                  </div>

                  {!modoRecuperarComerciante ? (
                    <div>
                      <label className="block text-xs font-bold text-neutral-700 mb-1">
                        Contraseña única
                      </label>
                      <div className="relative flex items-center">
                        <input
                          type={mostrarPasswordComerciante ? 'text' : 'password'}
                          required
                          value={passwordComerciante}
                          onChange={(e) => setPasswordComerciante(e.target.value)}
                          placeholder="••••••••"
                          className="w-full rounded-xl border border-neutral-200 bg-white pl-3.5 pr-11 py-2 text-sm focus:outline-none focus:border-orange-500"
                        />
                        <button
                          type="button"
                          onClick={() => setMostrarPasswordComerciante((prev) => !prev)}
                          aria-label="Mostrar u ocultar contraseña"
                          className="absolute right-2.5 text-neutral-500 hover:text-neutral-800 p-1 rounded-lg"
                        >
                          {mostrarPasswordComerciante ? '🙈' : '👁️'}
                        </button>
                      </div>
                    </div>
                  ) : (
                    <div>
                      <label className="block text-xs font-bold text-neutral-700 mb-1">
                        Nueva contraseña única
                      </label>
                      <div className="relative flex items-center">
                        <input
                          type={mostrarPasswordComerciante ? 'text' : 'password'}
                          required
                          value={nuevaPasswordComerciante}
                          onChange={(e) => setNuevaPasswordComerciante(e.target.value)}
                          placeholder="Escribí tu nueva contraseña"
                          className="w-full rounded-xl border border-neutral-200 bg-white pl-3.5 pr-11 py-2 text-sm focus:outline-none focus:border-orange-500"
                        />
                        <button
                          type="button"
                          onClick={() => setMostrarPasswordComerciante((prev) => !prev)}
                          aria-label="Mostrar u ocultar nueva contraseña"
                          className="absolute right-2.5 text-neutral-500 hover:text-neutral-800 p-1 rounded-lg"
                        >
                          {mostrarPasswordComerciante ? '🙈' : '👁️'}
                        </button>
                      </div>
                    </div>
                  )}

                  <div className="flex flex-wrap items-center justify-between gap-2 pt-1">
                    {!modoRegistroComerciante ? (
                      <button
                        type="button"
                        onClick={() => {
                          setModoRecuperarComerciante((prev) => !prev);
                          setIntentosFallidosComerciante(0);
                          setMensajeAuthComerciante('');
                          setErrorAuthComerciante('');
                        }}
                        className="text-xs font-bold text-orange-600 hover:underline"
                      >
                        {modoRecuperarComerciante
                          ? '← Volver a iniciar sesión'
                          : '¿Olvidaste tu contraseña? Recuperarla'}
                      </button>
                    ) : (
                      <span className="text-[11px] text-neutral-600">
                        Requiere aprobación del administrador tras registrarte.
                      </span>
                    )}

                    <button
                      type="submit"
                      className="bg-orange-600 hover:bg-orange-700 text-white text-xs sm:text-sm font-bold px-4 py-2 rounded-xl"
                    >
                      {modoRecuperarComerciante
                        ? 'Guardar nueva contraseña'
                        : modoRegistroComerciante
                          ? 'Crear cuenta'
                          : 'Ingresar'}
                    </button>
                  </div>
                </form>
              ) : (
                <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
                  Sesión activa como comerciante aprobado ({comercianteAutenticado}).
                </p>
              )}

              {errorAuthComerciante && (
                <p className="text-xs font-semibold text-red-700 bg-red-50 border border-red-200 rounded-xl px-3 py-2">
                  {errorAuthComerciante}
                </p>
              )}

              {mensajeAuthComerciante && !comercianteAutenticado && (
                <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
                  {mensajeAuthComerciante}
                </p>
              )}
            </div>

            {comercianteAutenticado && (
              <>
                <div className="flex items-center justify-between">
                  <p className="text-xs sm:text-sm text-neutral-600">
                    Administrá tus locales cargados en San Nicolás o agregá uno nuevo.
                  </p>
                  <button
                    type="button"
                    onClick={() => {
                      setPanelActivo(null);
                      setMostrarSumarLocal(true);
                    }}
                    className="bg-orange-600 hover:bg-orange-700 text-white text-xs font-bold px-3.5 py-2 rounded-xl shrink-0"
                  >
                    + Nuevo local
                  </button>
                </div>

                {comercios.length === 0 ? (
                  <p className="text-xs sm:text-sm text-neutral-500 text-center py-4 bg-neutral-50 rounded-xl border border-neutral-200">
                    Aún no hay comercios cargados desde este panel.
                  </p>
                ) : (
                  <div className="space-y-2.5">
                    {comercios.map((c) => (
                      <div
                        key={c.id}
                        className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 p-3 rounded-xl border border-neutral-200 bg-neutral-50"
                      >
                        <div>
                          <p className="text-sm font-extrabold text-neutral-900">{c.nombre}</p>
                          <p className="text-xs text-neutral-600">
                            {c.rubro} · {c.direccion}
                          </p>
                        </div>
                        <button
                          type="button"
                          onClick={() => eliminarComercio(c.id)}
                          className="text-xs font-bold px-3 py-1.5 rounded-lg bg-red-50 text-red-700 hover:bg-red-100"
                        >
                          Quitar
                        </button>
                      </div>
                    ))}
                  </div>
                )}
              </>
            )}
          </section>
        )}

        {panelActivo === 'admin' && (
          <section className="bg-white rounded-2xl border border-neutral-300 p-4 sm:p-5 space-y-4 shadow-sm">
            <div className="flex items-center justify-between border-b border-neutral-100 pb-3">
              <div>
                <span className="text-[11px] font-bold uppercase tracking-wider text-neutral-700 bg-neutral-100 px-2.5 py-0.5 rounded-full">
                  Control General
                </span>
                <h2 className="text-base sm:text-lg font-extrabold text-neutral-900 mt-1">
                  Panel de Administrador
                </h2>
              </div>
              <button
                type="button"
                onClick={() => setPanelActivo(null)}
                className="text-xs font-bold text-neutral-500 hover:text-neutral-800 px-2.5 py-1 rounded-lg bg-neutral-100"
              >
                Cerrar
              </button>
            </div>

            <div className="bg-neutral-50 border border-neutral-200 rounded-xl p-4 space-y-3">
              <div className="flex items-center justify-between gap-2">
                <h3 className="text-sm font-extrabold text-neutral-900">
                  {modoRecuperarAdmin
                    ? 'Recuperar contraseña de administrador'
                    : 'Acceso exclusivo de administrador'}
                </h3>
                {adminAutenticado && (
                  <button
                    type="button"
                    onClick={() => {
                      setAdminAutenticado(false);
                      setPasswordAdmin('');
                      setMensajeAuthAdmin('Sesión de administrador cerrada.');
                      setErrorAuthAdmin('');
                    }}
                    className="text-xs font-bold text-red-600 hover:underline"
                  >
                    Cerrar sesión
                  </button>
                )}
              </div>

              {!adminAutenticado ? (
                <form
                  onSubmit={(e) => {
                    e.preventDefault();
                    setErrorAuthAdmin('');
                    setMensajeAuthAdmin('');
                    const correo = emailAdmin.trim().toLowerCase();
                    if (correo !== ADMIN_EMAIL_UNICO) {
                      setErrorAuthAdmin(
                        'Acceso denegado: este panel está habilitado únicamente para la cuenta principal del administrador.'
                      );
                      return;
                    }
                    if (modoRecuperarAdmin) {
                      setMensajeAuthAdmin(
                        `Enviamos las instrucciones de recuperación al correo autorizado (${ADMIN_EMAIL_UNICO}).`
                      );
                      return;
                    }
                    if (!passwordAdmin.trim()) {
                      setErrorAuthAdmin('Ingresá la contraseña de administrador.');
                      return;
                    }
                    if (passwordAdmin !== ADMIN_PASSWORD_UNICA) {
                      setErrorAuthAdmin('Contraseña incorrecta para la cuenta de administrador.');
                      return;
                    }
                    setAdminAutenticado(true);
                    setCuentasComerciantes(leerCuentasComerciantes());
                    setMensajeAuthAdmin(
                      `Sesión iniciada como único administrador autorizado (${ADMIN_EMAIL_UNICO}).`
                    );
                  }}
                  className="space-y-3"
                >
                  <div>
                    <label className="block text-xs font-bold text-neutral-700 mb-1">
                      Correo de administrador
                    </label>
                    <input
                      type="email"
                      required
                      value={emailAdmin}
                      onChange={(e) => setEmailAdmin(e.target.value)}
                      placeholder="Correo autorizado del administrador"
                      className="w-full rounded-xl border border-neutral-200 bg-white px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
                    />
                  </div>

                  {!modoRecuperarAdmin && (
                    <div>
                      <label className="block text-xs font-bold text-neutral-700 mb-1">
                        Contraseña
                      </label>
                      <div className="relative flex items-center">
                        <input
                          type={mostrarPasswordAdmin ? 'text' : 'password'}
                          required
                          value={passwordAdmin}
                          onChange={(e) => setPasswordAdmin(e.target.value)}
                          placeholder="••••••••"
                          className="w-full rounded-xl border border-neutral-200 bg-white pl-3.5 pr-11 py-2 text-sm focus:outline-none focus:border-orange-500"
                        />
                        <button
                          type="button"
                          onClick={() => setMostrarPasswordAdmin((prev) => !prev)}
                          aria-label="Mostrar u ocultar contraseña de administrador"
                          className="absolute right-2.5 text-neutral-500 hover:text-neutral-800 p-1 rounded-lg"
                        >
                          {mostrarPasswordAdmin ? '🙈' : '👁️'}
                        </button>
                      </div>
                    </div>
                  )}

                  <div className="flex flex-wrap items-center justify-between gap-2 pt-1">
                    <button
                      type="button"
                      onClick={() => {
                        setModoRecuperarAdmin((prev) => !prev);
                        setMensajeAuthAdmin('');
                        setErrorAuthAdmin('');
                      }}
                      className="text-xs font-bold text-orange-600 hover:underline"
                    >
                      {modoRecuperarAdmin
                        ? '← Volver a iniciar sesión'
                        : '¿Olvidaste tu contraseña? Recuperarla'}
                    </button>

                    <button
                      type="submit"
                      className="bg-neutral-900 hover:bg-neutral-800 text-white text-xs sm:text-sm font-bold px-4 py-2 rounded-xl"
                    >
                      {modoRecuperarAdmin ? 'Enviar recuperación' : 'Ingresar'}
                    </button>
                  </div>
                </form>
              ) : (
                <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
                  Sesión activa como único administrador autorizado ({ADMIN_EMAIL_UNICO}).
                </p>
              )}

              {errorAuthAdmin && (
                <p className="text-xs font-semibold text-red-700 bg-red-50 border border-red-200 rounded-xl px-3 py-2">
                  {errorAuthAdmin}
                </p>
              )}

              {mensajeAuthAdmin && !adminAutenticado && (
                <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
                  {mensajeAuthAdmin}
                </p>
              )}
            </div>

            {adminAutenticado && (
              <div className="space-y-4">
                <div className="space-y-2.5">
                  <div className="flex items-center justify-between">
                    <h3 className="text-sm font-extrabold text-neutral-900">
                      Cuentas de comerciantes ({listaCuentasComerciantes.length})
                    </h3>
                  </div>

                  {listaCuentasComerciantes.length === 0 ? (
                    <p className="text-xs sm:text-sm text-neutral-500 text-center py-4 bg-neutral-50 rounded-xl border border-neutral-200">
                      Todavía no hay comerciantes registrados esperando aprobación.
                    </p>
                  ) : (
                    <div className="space-y-2">
                      {listaCuentasComerciantes.map(([correo, datos]) => (
                        <div
                          key={correo}
                          className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 p-3 rounded-xl border border-neutral-200 bg-neutral-50"
                        >
                          <div className="space-y-0.5">
                            <p className="text-sm font-extrabold text-neutral-900">{correo}</p>
                            <span
                              className={`inline-block text-[11px] font-bold px-2.5 py-0.5 rounded-full ${
                                datos.aprobado
                                  ? 'bg-emerald-100 text-emerald-800'
                                  : 'bg-amber-100 text-amber-800'
                              }`}
                            >
                              {datos.aprobado ? 'Aprobada' : 'Pendiente de aprobación'}
                            </span>
                          </div>
                          <div className="flex items-center gap-2">
                            {!datos.aprobado ? (
                              <button
                                type="button"
                                onClick={() => actualizarAprobacionComerciante(correo, true)}
                                className="text-xs font-bold px-3 py-1.5 rounded-lg bg-emerald-600 text-white hover:bg-emerald-700"
                              >
                                Aprobar cuenta
                              </button>
                            ) : (
                              <button
                                type="button"
                                onClick={() => actualizarAprobacionComerciante(correo, false)}
                                className="text-xs font-bold px-3 py-1.5 rounded-lg bg-amber-100 text-amber-900 hover:bg-amber-200"
                              >
                                Suspender
                              </button>
                            )}
                            <button
                              type="button"
                              onClick={() => eliminarCuentaComerciante(correo)}
                              className="text-xs font-bold px-3 py-1.5 rounded-lg bg-red-600 text-white hover:bg-red-700"
                            >
                              Eliminar
                            </button>
                          </div>
                        </div>
                      ))}
                    </div>
                  )}
                </div>

                <div className="space-y-2.5 pt-2 border-t border-neutral-100">
                  <div className="flex items-center justify-between">
                    <h3 className="text-sm font-extrabold text-neutral-900">
                      Comercios publicados ({comercios.length})
                    </h3>
                  </div>

                  {comercios.length === 0 ? (
                    <p className="text-xs sm:text-sm text-neutral-500 text-center py-4 bg-neutral-50 rounded-xl border border-neutral-200">
                      No hay comercios registrados para moderar todavía.
                    </p>
                  ) : (
                    <div className="space-y-2.5">
                      {comercios.map((c) => (
                        <div
                          key={c.id}
                          className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 p-3 rounded-xl border border-neutral-200 bg-neutral-50"
                        >
                          <div>
                            <p className="text-sm font-extrabold text-neutral-900">{c.nombre}</p>
                            <p className="text-xs text-neutral-500">
                              {c.rubro} · {c.direccion} · WhatsApp: {c.whatsapp}
                            </p>
                          </div>
                          <button
                            type="button"
                            onClick={() => eliminarComercio(c.id)}
                            className="text-xs font-bold px-3 py-1.5 rounded-lg bg-red-600 text-white hover:bg-red-700"
                          >
                            Eliminar
                          </button>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              </div>
            )}
          </section>
        )}

        <section id="seccion-categorias-sn" className="space-y-4">
          <div className="text-center space-y-1">
            <h2 className="text-xl sm:text-2xl font-extrabold text-neutral-900">Categorías</h2>
            <p className="text-xs sm:text-sm text-neutral-500">
              Elegí un rubro para ver productos y servicios disponibles
            </p>
            {rubroSeleccionado !== 'Todos' && (
              <div className="pt-1">
                <button
                  type="button"
                  onClick={() => setRubroSeleccionado('Todos')}
                  className="text-xs font-bold text-orange-600 hover:underline"
                >
                  Mostrando: {rubroSeleccionado} · Ver todas
                </button>
              </div>
            )}
          </div>

          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 sm:gap-4">
            {CATEGORIAS_CON_IMAGEN.map((cat) => {
              const activa = rubroSeleccionado === cat.nombre;
              return (
                <button
                  key={cat.nombre}
                  type="button"
                  onClick={() => setRubroSeleccionado(activa ? 'Todos' : cat.nombre)}
                  className={`bg-white rounded-xl border overflow-hidden text-center flex flex-col transition-transform hover:-translate-y-0.5 ${
                    activa
                      ? 'border-orange-500 ring-2 ring-orange-200'
                      : 'border-neutral-200 hover:border-orange-300'
                  }`}
                >
                  <div className="h-28 sm:h-32 w-full overflow-hidden bg-neutral-100">
                    <img
                      src={cat.imagen}
                      alt={cat.nombre}
                      className="w-full h-full object-cover"
                    />
                  </div>
                  <div className="p-2.5 sm:p-3 flex-1 flex items-center justify-center">
                    <span className="text-xs sm:text-sm font-bold text-neutral-900 leading-snug">
                      {cat.nombre}
                    </span>
                  </div>
                </button>
              );
            })}
          </div>
        </section>

        <section id="seccion-comercios-sn" className="space-y-3">
          <div className="flex items-center justify-between">
            <h2 className="text-lg font-extrabold text-neutral-900">Comercios disponibles</h2>
            <span className="text-xs font-semibold text-neutral-600">
              {comerciosFiltrados.length} {comerciosFiltrados.length === 1 ? 'local' : 'locales'}
            </span>
          </div>

          {comerciosFiltrados.length === 0 ? (
            <div className="bg-white rounded-2xl border border-neutral-200 p-6 text-center space-y-2">
              <p className="font-bold text-neutral-800">Todavía no hay comercios cargados</p>
              <p className="text-sm text-neutral-500">
                Usá el botón &ldquo;Sumá tu local&rdquo; para cargar el primer comercio y ubicarlo en el mapa.
              </p>
            </div>
          ) : (
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {comerciosFiltrados.map((comercio) => (
                <article
                  key={comercio.id}
                  onClick={() => {
                    setComercioActivoMapa(comercio);
                    setNuevaDireccion(comercio.direccion);
                    setMensajeEstado('');
                  }}
                  className="bg-white rounded-2xl border border-neutral-200 overflow-hidden flex flex-col justify-between cursor-pointer"
                >
                  <div>
                    <div className="h-44 w-full overflow-hidden bg-neutral-100">
                      <img
                        src={comercio.imagen}
                        alt={comercio.nombre}
                        className="w-full h-full object-cover"
                      />
                    </div>
                    <div className="p-4 space-y-2">
                      <div className="flex items-center justify-between gap-2">
                        <span className="text-[11px] font-bold uppercase tracking-wider text-orange-700 bg-orange-50 px-2.5 py-1 rounded-full">
                          {comercio.rubro}
                        </span>
                        <span className="text-xs font-semibold text-amber-700 bg-amber-50 px-2.5 py-0.5 rounded-full">
                          {comercio.promo}
                        </span>
                      </div>
                      <h3 className="text-base font-extrabold text-neutral-900">{comercio.nombre}</h3>
                      <p className="text-xs font-medium text-neutral-500">{comercio.direccion}</p>
                      <p className="text-sm text-neutral-600">{comercio.descripcion}</p>
                    </div>
                  </div>

                  <div className="px-4 py-3 border-t border-neutral-100 flex items-center justify-between gap-2">
                    <span className="text-xs font-bold text-orange-600">Ver en mapa</span>
                    <a
                      href={`https://wa.me/${comercio.whatsapp}?text=${encodeURIComponent(
                        `Hola ${comercio.nombre}, los vi en Mercado Nicoleño.`
                      )}`}
                      target="_blank"
                      rel="noreferrer"
                      onClick={(e) => e.stopPropagation()}
                      className="bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-bold px-3.5 py-2 rounded-lg"
                    >
                      WhatsApp
                    </a>
                  </div>
                </article>
              ))}
            </div>
          )}
        </section>

        <section
          id="seccion-mapa-san-nicolas"
          className="bg-white rounded-2xl border border-orange-100 p-4 sm:p-5 space-y-4"
        >
          <div>
            <h2 className="text-base sm:text-lg font-extrabold text-neutral-900">
              Ubicación del comercio en San Nicolás
            </h2>
            <p className="text-xs sm:text-sm text-neutral-600">
              {comercioActivoMapa ? (
                <>
                  Comercio:{' '}
                  <strong className="text-neutral-900">{comercioActivoMapa.nombre}</strong> —{' '}
                  <span className="text-orange-700 font-semibold">
                    {comercioActivoMapa.direccion}
                  </span>
                </>
              ) : (
                'Ingresá una dirección de San Nicolás para verla reflejada en el mapa.'
              )}
            </p>
          </div>

          <form onSubmit={actualizarDireccionMapa} className="flex flex-col sm:flex-row gap-2">
            <input
              type="text"
              value={nuevaDireccion}
              onChange={(e) => setNuevaDireccion(e.target.value)}
              placeholder="Ej.: Mitre 250, San Nicolás"
              className="flex-1 rounded-xl border border-neutral-200 px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
            />
            <button
              type="submit"
              className="bg-orange-600 hover:bg-orange-700 text-white text-xs sm:text-sm font-bold px-4 py-2 rounded-xl"
            >
              Actualizar dirección
            </button>
          </form>

          {mensajeEstado && (
            <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
              {mensajeEstado}
            </p>
          )}

          <div className="w-full h-64 sm:h-72 rounded-xl overflow-hidden border border-neutral-200 bg-neutral-100">
            <iframe
              title={`Mapa de ${comercioActivoMapa?.direccion || 'San Nicolás de los Arroyos'}`}
              src="https://www.openstreetmap.org/export/embed.html?bbox=-60.265%2C-33.365%2C-60.185%2C-33.315&layer=mapnik&marker=-33.3335%2C-60.2216"
              className="w-full h-full border-0"
              loading="lazy"
            />
          </div>
        </section>
      </main>
    </div>
  );
}

export default App;
