const esCorreoRealValido = (correo) => {
  const limpio = (correo || '').trim().toLowerCase();
  const match = limpio.match(/^[^\s@]+@([^\s@]+\.[^\s@]{2,})$/);
  if (!match) return false;
  const dominio = match[1];
  if (DOMINIOS_CORREO_FALSO.includes(dominio)) return false;
  return true;
};

const leerCuentasComerciantes = () => {
  try {
    const raw = JSON.parse(localStorage.getItem(COMERCIANTES_CUENTAS_KEY) || '{}');
    const normalizadas = {};
    Object.entries(raw).forEach(([correo, val]) => {
      if (typeof val === 'string') {
        normalizadas[correo] = {
          password: val,
          aprobado: true,
          creadoEn: new Date().toISOString()
        };
      } else if (val && typeof val === 'object') {
        normalizadas[correo] = {
          password: val.password || '',
          aprobado: Boolean(val.aprobado),
          creadoEn: val.creadoEn || new Date().toISOString()
        };
      }
    });
    return normalizadas;
  } catch {
    return {};
  }
};

const guardarCuentasComerciantes = (cuentas) => {

const construirUrlMapaSN = (direccion, nombreLugar) => {
  const dirLimpia = (direccion || '').trim();
  const nombreLimpio = (nombreLugar || '').trim();
  if (!dirLimpia && !nombreLimpio) {
    return 'https://maps.google.com/maps?q=San+Nicol%C3%A1s+de+los+Arroyos,+Buenos+Aires,+Argentina&t=m&z=14&output=embed';
  }
  const base = dirLimpia || nombreLimpio;
  const conCiudad = /san nicol[aá]s/i.test(base)
    ? `${base}, Buenos Aires, Argentina`
    : `${base}, San Nicolás de los Arroyos, Buenos Aires, Argentina`;
  const consulta =
    nombreLimpio && !conCiudad.toLowerCase().includes(nombreLimpio.toLowerCase())
      ? `${nombreLimpio}, ${conCiudad}`
      : conCiudad;
  return `https://maps.google.com/maps?q=${encodeURIComponent(consulta)}&t=m&z=17&output=embed`;
};

const formatearPrecioARS = (valor) => {
  const limpio = String(valor || '').trim();
  if (!limpio) return '';
  const numero = Number(limpio.replace(/[^0-9.,]/g, '').replace(',', '.'));
  if (!Number.isNaN(numero) && numero > 0) {
    return `$${numero.toLocaleString('es-AR')}`;
  }
  return limpio.startsWith('$') ? limpio : `$${limpio}`;
};

const DIAS_SEMANA_ES = ['domingo', 'lunes', 'martes', 'miércoles', 'jueves', 'viernes', 'sábado'];

const formatearFechaCortaSN = (fecha) => {
  const diaNombre = DIAS_SEMANA_ES[fecha.getDay()];
  const dd = String(fecha.getDate()).padStart(2, '0');
  const mm = String(fecha.getMonth() + 1).padStart(2, '0');
  return `${diaNombre} ${dd}/${mm}`;
};

// Mapea códigos meteorológicos WMO a ícono y descripción en español (solo °C)
const describirClimaWmoCelsius = (codigo) => {
  if (codigo === 0) return { icono: '☀️', estado: 'Soleado' };
  if (codigo === 1 || codigo === 2) return { icono: '🌤️', estado: 'Algo nublado' };
  if (codigo === 3) return { icono: '☁️', estado: 'Nublado' };
  if (codigo === 45 || codigo === 48) return { icono: '🌫️', estado: 'Neblina' };
  if ((codigo >= 51 && codigo <= 67) || (codigo >= 80 && codigo <= 82)) {
    return { icono: '🌧️', estado: 'Lluvia' };
  }
  if (codigo >= 95) return { icono: '⛈️', estado: 'Tormenta' };
  return { icono: '🌡️', estado: 'Despejado' };
};

// Respaldo diario/horario en grados Celsius para San Nicolás de los Arroyos (-33.3335, -60.211)
const obtenerClimaRespaldoCelsiusSN = (fecha = new Date()) => {
  const mmdd = `${String(fecha.getMonth() + 1).padStart(2, '0')}-${String(fecha.getDate()).padStart(2, '0')}`;
  const pronosticoDiario = {
    '11-28': { min: 17, max: 33, mat: 21, tar: 33, noc: 25, icono: '☀️', estado: 'Soleado' },
    '11-29': { min: 18, max: 33, mat: 23, tar: 33, noc: 25, icono: '☁️', estado: 'Nublado' },
    '11-30': { min: 19, max: 35, mat: 24, tar: 35, noc: 26, icono: '☀️', estado: 'Soleado' },
    '12-01': { min: 22, max: 27, mat: 23, tar: 27, noc: 22, icono: '🌧️', estado: 'Lluvia' },
    '12-02': { min: 17, max: 24, mat: 19, tar: 24, noc: 18, icono: '☁️', estado: 'Nublado' },
    '12-03': { min: 12, max: 25, mat: 17, tar: 25, noc: 17, icono: '🌤️', estado: 'Algo nublado' },
    '12-04': { min: 15, max: 28, mat: 20, tar: 28, noc: 21, icono: '🌤️', estado: 'Algo nublado' }
  };
  const base = pronosticoDiario[mmdd] || {
    min: 18,
    max: 30,
    mat: 22,
    tar: 30,
    noc: 23,
    icono: '☀️',
    estado: 'Soleado'
  };
  const hora = fecha.getHours();
  const tempC = hora < 11 ? base.mat : hora < 19 ? base.tar : base.noc;
  return {
    temperaturaC: tempC,
    icono: base.icono,
    estado: base.estado
  };
};

              <input

                placeholder="Dirección en San Nicolás (ej.: Mitre 250)"

              <input
                type="text"
                value={nuevoLocal.whatsapp}
                onChange={(e) => setNuevoLocal({ ...nuevoLocal, whatsapp: e.target.value })}
                placeholder="WhatsApp (ej.: 5493364000199)"
                className="rounded-xl border border-neutral-200 px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
              />
              <input
                type="text"
                value={nuevoLocal.descripcion}
                onChange={(e) => setNuevoLocal({ ...nuevoLocal, descripcion: e.target.value })}
                placeholder="Qué ofrecés (productos, horarios, servicios)"
                className="sm:col-span-2 rounded-xl border border-neutral-200 px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
              />

              <div className="sm:col-span-2 rounded-xl border border-orange-200 bg-orange-50/40 p-3.5 space-y-3">
                <p className="text-xs font-extrabold uppercase tracking-wide text-orange-900">
                  Foto de tu local / comercio (se muestra en el listado al buscar por rubro)
                </p>
                <div className="flex flex-wrap items-center gap-3">
                  <label className="rounded-xl border border-dashed border-orange-400 bg-white px-3.5 py-2 text-xs font-bold text-orange-900 hover:bg-orange-50 cursor-pointer flex items-center justify-between gap-2">

                <p className="text-xs font-extrabold uppercase tracking-wide text-amber-900">
                  Primer producto con foto y precio (opcional, también podés sumar más luego)
                </p>
                <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                  <input
                    type="text"
                    value={nuevoLocal.productoNombre}
                    onChange={(e) => setNuevoLocal({ ...nuevoLocal, productoNombre: e.target.value })}
                    placeholder="Nombre del producto (ej.: Milanesas de ternera x kg)"
                    className="rounded-xl border border-neutral-200 bg-white px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
                  />
                  <input
                    type="text"
                    value={nuevoLocal.productoPrecio}
                    onChange={(e) => setNuevoLocal({ ...nuevoLocal, productoPrecio: e.target.value })}
                    placeholder="Precio (ej.: 8500 o $8.500)"
                    className="rounded-xl border border-neutral-200 bg-white px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
                  />
                  <div className="flex items-center gap-2">
                    <label className="flex-1 rounded-xl border border-dashed border-amber-400 bg-white px-3.5 py-2 text-xs font-bold text-amber-900 hover:bg-amber-50 cursor-pointer flex items-center justify-between gap-2">
                      <span className="truncate">
                        {nuevoLocal.productoArchivoNombre
                          ? `Foto: ${nuevoLocal.productoArchivoNombre}`
                          : 'Subir foto desde tu dispositivo'}
                      </span>
                      <input
                        type="file"
                        accept="image/*"
                        onChange={manejarArchivoProductoNuevoLocal}
                        className="hidden"
                      />
                    </label>
                    {nuevoLocal.productoImagen && (
                      <button
                        type="button"
                        onClick={() =>
                          setNuevoLocal((prev) => ({
                            ...prev,
                            productoImagen: '',
                            productoArchivoNombre: ''
                          }))
                        }
                        className="text-xs font-bold text-red-700 bg-red-50 hover:bg-red-100 px-2.5 py-2 rounded-xl shrink-0"
                      >
                        Sacar foto
                      </button>
                    )}
                  </div>
                </div>
                {nuevoLocal.productoImagen && (
                  <div className="flex items-center gap-3 pt-1">
                    <img
                      src={nuevoLocal.productoImagen}
                      alt="Vista previa del producto"
                      className="w-14 h-14 rounded-lg object-cover border border-amber-200"
                    />
                    <div className="text-xs">
                      <p className="font-bold text-neutral-900">
                        {nuevoLocal.productoNombre.trim() || 'Producto'}
                      </p>
                      <p className="font-extrabold text-orange-700">
                        {formatearPrecioARS(nuevoLocal.productoPrecio) || 'Sin precio'}
                      </p>
                    </div>
                  </div>
                )}
              </div>

              <div className="sm:col-span-2 flex justify-end">
                <button
                  type="submit"
                  className="bg-orange-600 hover:bg-orange-700 text-white text-xs sm:text-sm font-bold px-5 py-2.5 rounded-xl"
                >
                  Guardar y ubicar en el mapa
                </button>
              </div>
            </form>
          </section>
        )}

        {panelActivo === 'comerciante' && (
          <section className="bg-white rounded-2xl border border-amber-300 p-4 sm:p-5 space-y-4 shadow-sm">
            <div className="flex items-center justify-between border-b border-neutral-100 pb-3">
              <div>
                <span className="text-[11px] font-bold uppercase tracking-wider text-amber-700 bg-amber-50 px-2.5 py-0.5 rounded-full">
                  Gestión de Local
                </span>
                <h2 className="text-base sm:text-lg font-extrabold text-neutral-900 mt-1">
                  Panel de Comerciante
                </h2>
              </div>
              <button
                type="button"
                onClick={() => setPanelActivo(null)}
                className="text-xs font-bold text-neutral-500 hover:text-neutral-800 px-2.5 py-1 rounded-lg bg-neutral-100"
              >
                Cerrar
              </button>
            </div>

            <div className="bg-amber-50/60 border border-amber-200 rounded-xl p-4 space-y-3">
              <div className="flex items-center justify-between gap-2">
                <h3 className="text-sm font-extrabold text-neutral-900">
                  {modoRecuperarComerciante
                    ? 'Recuperar contraseña de comerciante'
                    : modoRegistroComerciante
                      ? 'Crear cuenta única de comerciante'
                      : 'Ingresar a mi cuenta de comerciante'}
                </h3>
                {comercianteAutenticado && (
                  <button
                    type="button"
                    onClick={() => {
                      setComercianteAutenticado('');
                      guardarSesionComerciante('');
                      setPasswordComerciante('');
                      setMensajeAuthComerciante('Sesión de comerciante cerrada.');
                      setErrorAuthComerciante('');
                    }}
                    className="text-xs font-bold text-red-600 hover:underline"
                  >
                    Cerrar sesión
                  </button>
                )}
              </div>

              {!comercianteAutenticado && !modoRecuperarComerciante && (
                <div className="grid grid-cols-2 gap-1.5 bg-white p-1 rounded-xl border border-amber-200">
                  <button
                    type="button"
                    onClick={() => {
                      setModoRegistroComerciante(false);
                      setErrorAuthComerciante('');
                      setMensajeAuthComerciante('');
                      setIntentosFallidosComerciante(0);
                    }}
                    className={`py-1.5 text-xs font-bold rounded-lg transition ${
                      !modoRegistroComerciante
                        ? 'bg-orange-600 text-white'
                        : 'text-neutral-700 hover:bg-neutral-50'
                    }`}
                  >
                    Ya tengo cuenta
                  </button>
                  <button
                    type="button"
                    onClick={() => {
                      setModoRegistroComerciante(true);
                      setErrorAuthComerciante('');
                      setMensajeAuthComerciante('');
                      setIntentosFallidosComerciante(0);
                    }}
                    className={`py-1.5 text-xs font-bold rounded-lg transition ${
                      modoRegistroComerciante
                        ? 'bg-orange-600 text-white'
                        : 'text-neutral-700 hover:bg-neutral-50'
                    }`}
                  >
                    Crear cuenta nueva
                  </button>
                </div>
              )}

              {!comercianteAutenticado ? (
                <form
                  onSubmit={(e) => {
                    e.preventDefault();
                    setErrorAuthComerciante('');
                    setMensajeAuthComerciante('');
                    const correo = emailComerciante.trim().toLowerCase();
                    if (!correo) return;

                    if (correo === ADMIN_EMAIL_UNICO) {
                      setErrorAuthComerciante(
                        'Este correo corresponde al administrador. Ingresá desde el Panel de Administrador.'
                      );
                      return;
                    }

                    if (!esCorreoRealValido(correo)) {
                      setErrorAuthComerciante(
                        'Ingresá un correo electrónico real y válido (ej.: tunombre@gmail.com).'
                      );
                      return;
                    }

                    const cuentas = leerCuentasComerciantes();

                    if (modoRecuperarComerciante) {
                      if (!cuentas[correo]) {
                        setErrorAuthComerciante(
                          'No existe una cuenta registrada con ese correo de comerciante.'
                        );
                        return;
                      }
                      if (!nuevaPasswordComerciante.trim() || nuevaPasswordComerciante.trim().length < 4) {
                        setErrorAuthComerciante(
                          'Ingresá una nueva contraseña única de al menos 4 caracteres.'
                        );
                        return;
                      }
                      cuentas[correo] = {
                        ...cuentas[correo],
                        password: nuevaPasswordComerciante
                      };
                      guardarCuentasComerciantes(cuentas);
                      setCuentasComerciantes(cuentas);
                      setNuevaPasswordComerciante('');
                      setPasswordComerciante('');
                      setModoRecuperarComerciante(false);
                      setModoRegistroComerciante(false);
                      setMensajeAuthComerciante(
                        `Contraseña actualizada para ${correo}. Ya podés iniciar sesión con tu nueva clave.`
                      );
                      return;
                    }

                    if (!passwordComerciante.trim() || passwordComerciante.trim().length < 4) {
                      setErrorAuthComerciante(
                        'Ingresá tu contraseña única (mínimo 4 caracteres).'
                      );
                      return;
                    }

                    if (modoRegistroComerciante) {
                      if (cuentas[correo]) {
                        setErrorAuthComerciante(
                          'Ya existe una cuenta registrada con este correo. Cambiá a "Ya tengo cuenta" para ingresar.'
                        );
                        return;
                      }
                      cuentas[correo] = {
                        password: passwordComerciante,
                        aprobado: false,
                        creadoEn: new Date().toISOString()
                      };
                      guardarCuentasComerciantes(cuentas);
                      setCuentasComerciantes(cuentas);
                      setPasswordComerciante('');
                      setModoRegistroComerciante(false);

                        `Cuenta creada para ${correo}. Quedó pendiente de aprobación por el administrador antes de poder ingresar.`
                      );
                      return;
                    }

                    const cuenta = cuentas[correo];
                    if (!cuenta) {
                      setErrorAuthComerciante(
                        'Todavía no creaste una cuenta con este correo. Tocá "Crear cuenta nueva" para registrarte.'
                      );
                      return;
                    }

                    if (cuenta.password !== passwordComerciante) {
                      const nuevoIntento = intentosFallidosComerciante + 1;
                      setIntentosFallidosComerciante(nuevoIntento);
                      setErrorAuthComerciante(
                        `Contraseña incorrecta (intento ${nuevoIntento}). Podés volver a intentarlo las veces que quieras o elegir "¿Olvidaste tu contraseña? Recuperarla".`
                      );
                      return;
                    }

                    if (!cuenta.aprobado) {
                      setIntentosFallidosComerciante(0);
                      setErrorAuthComerciante(
                        `Tu cuenta (${correo}) ya fue creada, pero todavía está pendiente de aprobación por el administrador.`
                      );
                      return;
                    }

                    setIntentosFallidosComerciante(0);
                    setComercianteAutenticado(correo);
                    guardarSesionComerciante(correo);
                    setMensajeAuthComerciante(`Sesión iniciada como comerciante aprobado (${correo}).`);
                  }}
                  className="space-y-3"
                >
                  <div>
                    <label className="block text-xs font-bold text-neutral-700 mb-1">
                      Correo electrónico real
                    </label>
                    <input
                      type="email"
                      required
                      value={emailComerciante}
                      onChange={(e) => setEmailComerciante(e.target.value)}
                      placeholder="tucomercio@gmail.com"
                      className="w-full rounded-xl border border-neutral-200 bg-white px-3.5 py-2 text-sm focus:outline-none focus:border-orange-500"
                    />
                  </div>

                  {!modoRecuperarComerciante ? (
                    <div>
                      <label className="block text-xs font-bold text-neutral-700 mb-1">
                        Contraseña única
                      </label>
                      <div className="relative flex items-center">
                        <input
                          type={mostrarPasswordComerciante ? 'text' : 'password'}
                          required
                          value={passwordComerciante}
                          onChange={(e) => setPasswordComerciante(e.target.value)}
                          placeholder="••••••••"
                          className="w-full rounded-xl border border-neutral-200 bg-white pl-3.5 pr-11 py-2 text-sm focus:outline-none focus:border-orange-500"
                        />
                        <button
                          type="button"
                          onClick={() => setMostrarPasswordComerciante((prev) => !prev)}
                          aria-label="Mostrar u ocultar contraseña"
                          className="absolute right-2.5 text-neutral-500 hover:text-neutral-800 p-1 rounded-lg"
                        >
                          {mostrarPasswordComerciante ? '🙈' : '👁️'}
                        </button>
                      </div>
                    </div>
                  ) : (
                    <div>
                      <label className="block text-xs font-bold text-neutral-700 mb-1">
                        Nueva contraseña única
                      </label>
                      <div className="relative flex items-center">
                        <input
                          type={mostrarPasswordComerciante ? 'text' : 'password'}
                          required
                          value={nuevaPasswordComerciante}
                          onChange={(e) => setNuevaPasswordComerciante(e.target.value)}
                          placeholder="Escribí tu nueva contraseña"
                          className="w-full rounded-xl border border-neutral-200 bg-white pl-3.5 pr-11 py-2 text-sm focus:outline-none focus:border-orange-500"
                        />
                        <button
                          type="button"
                          onClick={() => setMostrarPasswordComerciante((prev) => !prev)}
                          aria-label="Mostrar u ocultar nueva contraseña"
                          className="absolute right-2.5 text-neutral-500 hover:text-neutral-800 p-1 rounded-lg"
                        >
                          {mostrarPasswordComerciante ? '🙈' : '👁️'}
                        </button>
                      </div>
                    </div>
                  )}

                  <div className="flex flex-wrap items-center justify-between gap-2 pt-1">
                    {!modoRegistroComerciante ? (
                      <button
                        type="button"
                        onClick={() => {
                          setModoRecuperarComerciante((prev) => !prev);
                          setIntentosFallidosComerciante(0);
                          setMensajeAuthComerciante('');
                          setErrorAuthComerciante('');
                        }}
                        className="text-xs font-bold text-orange-600 hover:underline"
                      >
                        {modoRecuperarComerciante
                          ? '← Volver a iniciar sesión'
                          : '¿Olvidaste tu contraseña? Recuperarla'}
                      </button>
                    ) : (
                      <span className="text-[11px] text-neutral-600">
                        Requiere aprobación del administrador tras registrarte.
                      </span>
                    )}

                    <button
                      type="submit"
                      className="bg-orange-600 hover:bg-orange-700 text-white text-xs sm:text-sm font-bold px-4 py-2 rounded-xl"
                    >
                      {modoRecuperarComerciante
                        ? 'Guardar nueva contraseña'
                        : modoRegistroComerciante
                          ? 'Crear cuenta'
                          : 'Ingresar'}
                    </button>
                  </div>
                </form>
              ) : (
                <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
                  Sesión activa como comerciante aprobado ({comercianteAutenticado}).
                </p>
              )}

              {errorAuthComerciante && (
                <p className="text-xs font-semibold text-red-700 bg-red-50 border border-red-200 rounded-xl px-3 py-2">
                  {errorAuthComerciante}
                </p>
              )}

              {mensajeAuthComerciante && !comercianteAutenticado && (
                <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
                  {mensajeAuthComerciante}
                </p>
              )}
            </div>

            {comercianteAutenticado && (
              <>
                <div className="flex items-center justify-between">
                  <p className="text-xs sm:text-sm text-neutral-600">
                    Administrá tus locales cargados en San Nicolás, agregá productos con foto y precio o agregá uno nuevo.
                  </p>
                  <button
                    type="button"
                    onClick={() => {
                      setPanelActivo(null);
                      setMostrarSumarLocal(true);
                    }}
                    className="bg-orange-600 hover:bg-orange-700 text-white text-xs font-bold px-3.5 py-2 rounded-xl shrink-0"
                  >
                    + Nuevo local
                  </button>
                </div>

                {comercios.length === 0 ? (
                  <p className="text-xs sm:text-sm text-neutral-500 text-center py-4 bg-neutral-50 rounded-xl border border-neutral-200">
                    Aún no hay comercios cargados desde este panel.
                  </p>
                ) : (
                  <div className="space-y-3">
                    {comercios.map((c) => {
                      const prodDraft = nuevoProductoPorComercio[c.id] || {
                        nombre: '',
                        precio: '',
                        imagen: '',
                        archivoNombre: ''
                      };
                      const comercioDraft = obtenerBorradorEdicionComercio(c);
                      return (
                        <div
                          key={c.id}
                          className="p-3.5 rounded-xl border border-neutral-200 bg-neutral-50 space-y-3"
                        >
                          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
                            <div>
                              <p className="text-sm font-extrabold text-neutral-900">{c.nombre}</p>
                              <p className="text-xs text-neutral-600">
                                {c.rubro} · {c.direccion}
                              </p>
                            </div>
                            <div className="flex items-center gap-2">
                              <button
                                type="button"
                                onClick={() => {
                                  setComercioActivoMapa(c);
                                  setMensajeEstado(`Ubicación del local: ${c.nombre} (${c.direccion})`);
                                  irAMapaYCercania();
                                }}
                                className="text-xs font-bold px-3 py-1.5 rounded-lg bg-orange-50 text-orange-700 hover:bg-orange-100"
                              >
                                Ver en mapa
                              </button>
                              <button
                                type="button"
                                onClick={() => eliminarComercio(c.id)}
                                className="text-xs font-bold px-3 py-1.5 rounded-lg bg-red-50 text-red-700 hover:bg-red-100"
                              >
                                Quitar
                              </button>
                            </div>
                          </div>

                          <div className="rounded-xl border border-orange-200 bg-white p-3 space-y-2.5">
                            <p className="text-[11px] font-extrabold uppercase tracking-wide text-orange-900">
                              Editar datos generales y foto de portada del negocio
                            </p>
                            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                              <input
                                type="text"
                                value={comercioDraft.nombre}
                                onChange={(e) =>
                                  actualizarCampoEdicionComercio(c, 'nombre', e.target.value)
                                }
                                placeholder="Nombre del comercio"
                                className="rounded-lg border border-neutral-200 bg-neutral-50 px-3 py-1.5 text-xs focus:outline-none focus:border-orange-500"
                              />
                              <select
                                value={comercioDraft.rubro}
                                onChange={(e) =>
                                  actualizarCampoEdicionComercio(c, 'rubro', e.target.value)
                                }
                                className="rounded-lg border border-neutral-200 bg-neutral-50 px-3 py-1.5 text-xs focus:outline-none focus:border-orange-500"
                              >
                                {CATEGORIAS_CON_IMAGEN.filter((cat) => !cat.soloTurno).map((cat) => (
                                  <option key={cat.nombre} value={cat.nombre}>
                                    {cat.nombre}
                                  </option>
                                ))}
                              </select>
                              <input
                                type="text"
                                value={comercioDraft.direccion}
                                onChange={(e) =>
                                  actualizarCampoEdicionComercio(c, 'direccion', e.target.value)
                                }
                                placeholder="Dirección en San Nicolás (ej.: Mitre 250)"
                                className="rounded-lg border border-neutral-200 bg-neutral-50 px-3 py-1.5 text-xs focus:outline-none focus:border-orange-500"
                              />
                              <input
                                type="text"
                                value={comercioDraft.whatsapp}
                                onChange={(e) =>
                                  actualizarCampoEdicionComercio(c, 'whatsapp', e.target.value)
                                }
                                placeholder="WhatsApp (ej.: 5493364000199)"
                                className="rounded-lg border border-neutral-200 bg-neutral-50 px-3 py-1.5 text-xs focus:outline-none focus:border-orange-500"
                              />
                              <input
                                type="text"
                                value={comercioDraft.descripcion}
                                onChange={(e) =>
                                  actualizarCampoEdicionComercio(c, 'descripcion', e.target.value)
                                }
                                placeholder="Descripción del comercio (horarios, servicios, etc.)"
                                className="sm:col-span-2 rounded-lg border border-neutral-200 bg-neutral-50 px-3 py-1.5 text-xs focus:outline-none focus:border-orange-500"
                              />
                            </div>

                            <div className="flex flex-wrap items-center justify-between gap-2 pt-1">
                              <div className="flex items-center gap-2.5">
                                {comercioDraft.imagen && (
                                  <img
                                    src={comercioDraft.imagen}
                                    alt={`Portada de ${comercioDraft.nombre || c.nombre}`}
                                    className="w-14 h-11 rounded-lg object-cover border border-orange-200 shrink-0"
                                  />
                                )}
                                <label className="rounded-lg border border-dashed border-orange-400 bg-orange-50/50 px-3 py-1.5 text-[11px] font-bold text-orange-900 hover:bg-orange-100/70 cursor-pointer">
                                  <span>
                                    {comercioDraft.archivoNombre
                                      ? `Portada: ${comercioDraft.archivoNombre}`
                                      : 'Cambiar foto de portada'}
                                  </span>
                                  <input

                {farmaciasTurnoHoy.siguiente.etiqueta}
              </h3>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                {farmaciasTurnoHoy.siguiente.farmacias.map((f) => (
                  <div
                    key={f.nombre}
                    className="rounded-xl border border-neutral-200 bg-neutral-50 p-3.5 flex items-center justify-between gap-3"
                  >
                    <div>
                      <p className="text-sm font-extrabold text-neutral-900">{f.nombre}</p>
                      <p className="text-xs text-neutral-600">{f.direccion}</p>
                    </div>
                    <button
                      type="button"
                      onClick={() => {
                        setComercioActivoMapa({
                          nombre: f.nombre,
                          direccion: f.direccion
                        });
                        setMensajeEstado(`Farmacia de turno ubicada en el mapa: ${f.nombre} (${f.direccion})`);
                        irAMapaYCercania();
                      }}
                      className="text-xs font-bold text-orange-700 bg-white hover:bg-orange-50 border border-neutral-200 px-3 py-1.5 rounded-lg shrink-0"
                    >
                      Ver en mapa
                    </button>
                  </div>
                ))}
              </div>
            </div>
          </section>
        ) : (
          <section id="seccion-comercios-sn" className="space-y-3">
            <div className="flex items-center justify-between">
              <h2 className="text-lg font-extrabold text-neutral-900">Comercios disponibles</h2>
              <span className="text-xs font-semibold text-neutral-600">
                {comerciosFiltrados.length} {comerciosFiltrados.length === 1 ? 'local' : 'locales'}
              </span>
            </div>

            {comerciosFiltrados.length === 0 ? (
              <div className="bg-white rounded-2xl border border-neutral-200 p-6 text-center space-y-2">
                <p className="font-bold text-neutral-800">Todavía no hay comercios cargados</p>
                <p className="text-sm text-neutral-500">
                  Usá el botón &ldquo;Sumá tu local&rdquo; para cargar el primer comercio y ubicarlo en el mapa.
                </p>
              </div>
            ) : comercioSeleccionadoDetalle ? (
              <div className="bg-white rounded-2xl border border-orange-200 overflow-hidden shadow-sm space-y-4">
                <div className="px-4 py-3 bg-orange-50/60 border-b border-orange-100 flex items-center justify-between gap-2">
                  <button
                    type="button"
                    onClick={() => setComercioSeleccionadoDetalle(null)}
                    className="text-xs font-bold text-orange-700 hover:underline"
                  >
                    ← Volver a los comercios
                  </button>
                  <span className="text-[11px] font-bold uppercase tracking-wider text-orange-700 bg-white px-2.5 py-0.5 rounded-full border border-orange-200">
                    {comercioSeleccionadoDetalle.rubro}
                  </span>
                </div>

                <div className="px-4 sm:px-5 space-y-4 pb-5">
                  <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 items-center">
                    <div className="h-44 sm:h-40 w-full rounded-xl overflow-hidden bg-neutral-100 border border-neutral-200">
                      <img
                        src={
                          comercioSeleccionadoDetalle.imagen ||
                          CATEGORIAS_CON_IMAGEN.find(
                            (c) => c.nombre === comercioSeleccionadoDetalle.rubro
                          )?.imagen ||
                          CATEGORIAS_CON_IMAGEN[0].imagen
                        }
                        alt={comercioSeleccionadoDetalle.nombre}
                        className="w-full h-full object-cover"
                      />
                    </div>
                    <div className="sm:col-span-2 space-y-2">
                      <h3 className="text-lg sm:text-xl font-extrabold text-neutral-900">
                        {comercioSeleccionadoDetalle.nombre}
                      </h3>
                      <p className="text-xs sm:text-sm font-semibold text-neutral-600">
                        {comercioSeleccionadoDetalle.direccion}
                      </p>
                      <p className="text-sm text-neutral-600">
                        {comercioSeleccionadoDetalle.descripcion}
                      </p>
                      <div className="pt-1 flex flex-wrap items-center gap-3">
                        <button
                          type="button"
                          onClick={() => {
                            setComercioActivoMapa(comercioSeleccionadoDetalle);
                            setMensajeEstado(
                              `Ubicación del comercio: ${comercioSeleccionadoDetalle.nombre} — ${comercioSeleccionadoDetalle.direccion}`
                            );
                            irAMapaYCercania();
                          }}
                          className="text-xs font-bold text-orange-600 hover:underline"
                        >
                          Ver en mapa
                        </button>
                        <a
                          href={`https://wa.me/${comercioSeleccionadoDetalle.whatsapp}?text=${encodeURIComponent(
                            `Hola ${comercioSeleccionadoDetalle.nombre}, los vi en Mercado Nicoleño.`
                          )}`}
                          target="_blank"
                          rel="noreferrer"
                          className="bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-bold px-3.5 py-2 rounded-lg"
                        >
                          WhatsApp
                        </a>
                      </div>
                    </div>
                  </div>

                  <div className="pt-3 border-t border-neutral-100 space-y-3">
                    <h4 className="text-sm font-extrabold text-neutral-900">
                      Productos del local
                    </h4>
                    {(comercioSeleccionadoDetalle.productos || []).length === 0 ? (
                      <p className="text-xs sm:text-sm text-neutral-500 bg-neutral-50 rounded-xl p-4 border border-neutral-200 text-center">
                        Este comercio aún no cargó productos en su catálogo.
                      </p>
                    ) : (
                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                        {(comercioSeleccionadoDetalle.productos || []).map((prod) => (
                          <div
                            key={prod.id}
                            className="flex items-center gap-3 rounded-xl border border-neutral-200 bg-neutral-50 p-2.5"
                          >
                            {prod.imagen && (
                              <img
                                src={prod.imagen}
                                alt={prod.nombre}
                                className="w-16 h-16 rounded-lg object-cover shrink-0 bg-white border border-neutral-200"
                              />
                            )}
                            <div className="min-w-0">
                              <p className="text-sm font-extrabold text-neutral-900 truncate">
                                {prod.nombre}
                              </p>
                              <p className="text-sm font-extrabold text-orange-700">
                                {prod.precio}
                              </p>
                            </div>
                          </div>
                        ))}
                      </div>
                    )}
                  </div>
                </div>
              </div>
            ) : (
              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                {comerciosFiltrados.map((comercio) => {
                  const fotoLocal =
                    comercio.imagen && !String(comercio.imagen).includes('leaflet')
                      ? comercio.imagen
                      : CATEGORIAS_CON_IMAGEN.find((c) => c.nombre === comercio.rubro)?.imagen ||
                        CATEGORIAS_CON_IMAGEN[0].imagen;
                  return (
                    <article
                      key={comercio.id}
                      onClick={() => {
                        setComercioSeleccionadoDetalle(comercio);
                        setComercioActivoMapa(comercio);
                        setMensajeEstado('');
                      }}
                      className="bg-white rounded-2xl border border-neutral-200 overflow-hidden flex flex-col justify-between cursor-pointer hover:border-orange-300 transition"
                    >
                      <div>
                        <div className="h-44 w-full overflow-hidden bg-neutral-100">
                          <img
                            src={fotoLocal}
                            alt={comercio.nombre}
                            className="w-full h-full object-cover"
                          />
                        </div>
                        <div className="p-4 space-y-2">
                          <span className="inline-block text-[11px] font-bold uppercase tracking-wider text-orange-700 bg-orange-50 px-2.5 py-1 rounded-full">
                            {comercio.rubro}
                          </span>
                          <h3 className="text-base font-extrabold text-neutral-900">{comercio.nombre}</h3>
                          <p className="text-xs font-medium text-neutral-500">{comercio.direccion}</p>
                          <p className="text-sm text-neutral-600">{comercio.descripcion}</p>
                        </div>
                      </div>

                      <div className="px-4 py-3 border-t border-neutral-100 flex items-center justify-between gap-2">
                        <button
                          type="button"
                          onClick={(e) => {
                            e.stopPropagation();
                            setComercioActivoMapa(comercio);
                            setMensajeEstado(
                              `Ubicación del comercio: ${comercio.nombre} — ${comercio.direccion}`
                            );
                            irAMapaYCercania();
                          }}
                          className="text-xs font-bold text-orange-600 hover:underline"
                        >
                          Ver en mapa
                        </button>
                        <button
                          type="button"
                          onClick={(e) => {
                            e.stopPropagation();
                            setComercioSeleccionadoDetalle(comercio);
                            setComercioActivoMapa(comercio);
                          }}
                          className="bg-orange-600 hover:bg-orange-700 text-white text-xs font-bold px-3.5 py-2 rounded-lg"
                        >
                          Ver productos
                        </button>
                      </div>
                    </article>
                  );
                })}
              </div>
            )}
          </section>
        )}

        <section
          id="seccion-mapa-san-nicolas"
          className="bg-white rounded-2xl border border-orange-100 p-4 sm:p-5 space-y-4"
        >
          <div>
            <h2 className="text-base sm:text-lg font-extrabold text-neutral-900">
              Ubicación en San Nicolás
            </h2>
            <p className="text-xs sm:text-sm text-neutral-600">
              {comercioActivoMapa ? (
                <>
                  Ubicación:{' '}
                  <strong className="text-neutral-900">{comercioActivoMapa.nombre}</strong> —{' '}
                  <span className="text-orange-700 font-semibold">
                    {comercioActivoMapa.direccion}
                  </span>
                </>
              ) : (
                'Tocá "Ver en mapa" en cualquier comercio o farmacia de turno para ver su ubicación.'
              )}
            </p>
          </div>

          {mensajeEstado && (
            <p className="text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2">
              {mensajeEstado}
            </p>
          )}

          <div className="w-full h-64 sm:h-72 rounded-xl overflow-hidden border border-neutral-200 bg-neutral-100">
            <iframe
              key={`${comercioActivoMapa?.direccion || 'centro'}-${comercioActivoMapa?.nombre || ''}`}
              title={`Mapa de ${comercioActivoMapa?.direccion || 'San Nicolás de los Arroyos'}`}
              src={construirUrlMapaSN(comercioActivoMapa?.direccion, comercioActivoMapa?.nombre)}
              className="w-full h-full border-0"
              loading="lazy"
            />
          </div>
        </section>
      </main>
    </div>
  );
}

export default App;
